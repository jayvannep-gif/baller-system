function renderUserProfileCard(data) { const container = document.getElementById('user-profile-display'); if (!container) return; // Payment Badge Status let paymentBadge = `N/A`; if (data.status === 'in') { paymentBadge = data.isPaid ? `PAID` : `UNPAID`; } // Warning Banner HTML Component const warningHtml = data.warning ? `
⚠️

ACCOUNT WARNING NOTICE

${data.warning}

` : ''; container.innerHTML = ` ${warningHtml}

${data.name}

${data.position} | ${data.height || "N/A"}

${paymentBadge}
`; }
